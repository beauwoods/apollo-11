# t4_lockout
################################################################
1. uplink VERB 1 6      (well-formed -- should be accepted)
   VERB field = 16

2. uplink malformed word 77777  (fails triple-redundancy -> TMFAIL2)
   VERB field = 16

3. uplink VERB 2 1      (should now be IGNORED -- UPLOCKFL set)
   VERB field = 16   <- unchanged means uplink is locked out

4. KEYBOARD VERB 2 1    (same keystrokes, physical panel)
   VERB field = 21   <- keyboard still works; block is uplink-specific

5. uplink ERROR RESET   (ELRCODE octal 22 -- unauthenticated, clears the lock)
6. uplink VERB 3 5      (uplink restored?)
   VERB field = 35   <- changed means the attacker unlocked themselves
